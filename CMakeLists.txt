cmake_minimum_required(VERSION 3.16)
project(Hail VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)

option(HA_BUILD_SHARED "Build shared Hail library" OFF)
option(HA_BUILD_TESTS "Build portable conformance tests" ON)
option(HA_BUILD_EXAMPLES "Build virtual audio examples" ON)
option(HA_BACKEND_VIRTUAL "Build externally driven deterministic backend" ON)
option(HA_FREESTANDING "Exclude hosted allocation helpers" OFF)
option(HA_BUILD_FUZZER "Build Clang libFuzzer config/virtual target" OFF)
option(HA_BACKEND_COREAUDIO "Build CoreAudio backend (not implemented yet)" OFF)
option(HA_BACKEND_WASAPI "Build WASAPI backend (not implemented yet)" OFF)
option(HA_BACKEND_PIPEWIRE "Build PipeWire backend (not implemented yet)" OFF)
option(HA_BACKEND_ALSA "Build ALSA backend (not implemented yet)" OFF)
set(HA_SANITIZER "" CACHE STRING "address, thread, or undefined (GCC/Clang)")
foreach(backend COREAUDIO WASAPI PIPEWIRE ALSA)
  if(HA_BACKEND_${backend})
    message(FATAL_ERROR "${backend} is a Gate D backend and is not implemented")
  endif()
endforeach()
set(HA_KIND STATIC)
if(HA_BUILD_SHARED)
  set(HA_KIND SHARED)
endif()
add_library(hail ${HA_KIND} src/core.c src/queue.c)
add_library(Hail::hail ALIAS hail)
target_compile_features(hail PRIVATE c_std_11)
set_target_properties(hail PROPERTIES C_EXTENSIONS OFF C_VISIBILITY_PRESET hidden
  VERSION ${PROJECT_VERSION} SOVERSION 0)
target_include_directories(hail PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_definitions(hail PRIVATE HA_BUILDING_LIBRARY
  HA_ENABLE_VIRTUAL=$<BOOL:${HA_BACKEND_VIRTUAL}> HA_FREESTANDING=$<BOOL:${HA_FREESTANDING}>)
if(NOT HA_BUILD_SHARED)
  target_compile_definitions(hail PUBLIC HA_STATIC)
endif()
if(NOT HA_FREESTANDING)
  target_sources(hail PRIVATE src/hosted.c)
endif()
if(HA_BACKEND_VIRTUAL)
  target_sources(hail PRIVATE src/virtual.c)
endif()
if(MSVC)
  target_compile_options(hail PRIVATE /W4 /WX)
else()
  target_compile_options(hail PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow)
endif()
if(HA_SANITIZER)
  if(MSVC)
    message(FATAL_ERROR "Use a GCC/Clang toolchain for HA_SANITIZER")
  endif()
  if(HA_SANITIZER STREQUAL "address")
    set(HA_SAN_FLAGS -fsanitize=address,undefined -fno-omit-frame-pointer)
  elseif(HA_SANITIZER STREQUAL "thread")
    set(HA_SAN_FLAGS -fsanitize=thread -fno-omit-frame-pointer)
  elseif(HA_SANITIZER STREQUAL "undefined")
    set(HA_SAN_FLAGS -fsanitize=undefined -fno-omit-frame-pointer)
  else()
    message(FATAL_ERROR "Unknown HA_SANITIZER")
  endif()
  target_compile_options(hail PUBLIC ${HA_SAN_FLAGS})
  target_link_options(hail PUBLIC ${HA_SAN_FLAGS})
endif()

file(STRINGS cmake/exports.txt HA_EXPORTS)
if(HA_BACKEND_VIRTUAL)
  list(APPEND HA_EXPORTS ha_virtual_configure ha_virtual_step ha_virtual_device_set
    ha_virtual_device_add ha_virtual_default_set)
endif()
if(HA_BUILD_SHARED)
  if(WIN32)
    set(HA_EXPORT_TEXT "EXPORTS\n")
    foreach(symbol IN LISTS HA_EXPORTS)
      string(APPEND HA_EXPORT_TEXT "${symbol}\n")
    endforeach()
    file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/hail.def "${HA_EXPORT_TEXT}")
    target_sources(hail PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/hail.def)
  elseif(APPLE)
    set(HA_EXPORT_TEXT "")
    foreach(symbol IN LISTS HA_EXPORTS)
      string(APPEND HA_EXPORT_TEXT "_${symbol}\n")
    endforeach()
    file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/hail.exports "${HA_EXPORT_TEXT}")
    target_link_options(hail PRIVATE "LINKER:-exported_symbols_list,${CMAKE_CURRENT_BINARY_DIR}/hail.exports")
  else()
    set(HA_EXPORT_TEXT "HAIL_0 { global:\n")
    foreach(symbol IN LISTS HA_EXPORTS)
      string(APPEND HA_EXPORT_TEXT "  ${symbol};\n")
    endforeach()
    string(APPEND HA_EXPORT_TEXT "local: *; };\n")
    file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/hail.map "${HA_EXPORT_TEXT}")
    target_link_options(hail PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_BINARY_DIR}/hail.map")
  endif()
endif()

if(HA_BUILD_TESTS)
  enable_language(CXX)
  enable_testing()
  add_executable(hail_headers tests/headers.cpp)
  target_compile_features(hail_headers PRIVATE cxx_std_11)
  target_link_libraries(hail_headers PRIVATE Hail::hail)
  add_test(NAME headers_cpp COMMAND hail_headers)
  if(HA_BACKEND_VIRTUAL)
    add_executable(hail_conformance tests/conformance.c)
    target_include_directories(hail_conformance PRIVATE src)
    target_compile_definitions(hail_conformance PRIVATE
      HA_ENABLE_VIRTUAL=1 HA_FREESTANDING=$<BOOL:${HA_FREESTANDING}>)
    target_link_libraries(hail_conformance PRIVATE Hail::hail)
    target_compile_features(hail_conformance PRIVATE c_std_11)
    add_test(NAME conformance COMMAND hail_conformance)
    find_package(Threads REQUIRED)
    add_executable(hail_concurrency tests/concurrency.cpp)
    target_link_libraries(hail_concurrency PRIVATE Hail::hail Threads::Threads)
    target_compile_features(hail_concurrency PRIVATE cxx_std_11)
    add_test(NAME concurrency COMMAND hail_concurrency)
  endif()
endif()
if(HA_BUILD_EXAMPLES AND HA_BACKEND_VIRTUAL)
  add_executable(hail_virtual_example examples/virtual.c)
  target_compile_features(hail_virtual_example PRIVATE c_std_11)
  target_link_libraries(hail_virtual_example PRIVATE Hail::hail)
  if(HA_BUILD_TESTS)
    add_test(NAME example COMMAND hail_virtual_example)
  endif()
endif()
if(HA_BUILD_FUZZER)
  if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang" OR NOT HA_BACKEND_VIRTUAL OR HA_BUILD_SHARED)
    message(FATAL_ERROR "Fuzzer needs Clang, a static library and the virtual backend")
  endif()
  if(HA_BUILD_TESTS OR HA_BUILD_EXAMPLES)
    message(FATAL_ERROR "Use a separate fuzzer build with HA_BUILD_TESTS=OFF and HA_BUILD_EXAMPLES=OFF")
  endif()
  target_compile_options(hail PRIVATE -fsanitize=fuzzer-no-link)
  add_executable(hail_fuzz_config fuzz/config.c)
  target_compile_features(hail_fuzz_config PRIVATE c_std_11)
  target_compile_options(hail_fuzz_config PRIVATE -fsanitize=fuzzer)
  target_link_options(hail_fuzz_config PRIVATE -fsanitize=fuzzer)
  target_link_libraries(hail_fuzz_config PRIVATE Hail::hail)
endif()

install(TARGETS hail EXPORT HailTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/hail.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
if(HA_BACKEND_VIRTUAL)
  install(FILES include/hail/virtual.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/hail)
endif()
install(EXPORT HailTargets NAMESPACE Hail:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Hail)
configure_package_config_file(cmake/HailConfig.cmake.in
  ${CMAKE_CURRENT_BINARY_DIR}/HailConfig.cmake
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Hail)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/HailConfigVersion.cmake
  VERSION ${PROJECT_VERSION} COMPATIBILITY ExactVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/HailConfig.cmake
  ${CMAKE_CURRENT_BINARY_DIR}/HailConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Hail)
set(HA_PC_PRIVATE "")
if(NOT HA_BUILD_SHARED)
  set(HA_PC_PRIVATE "-DHA_STATIC")
endif()
file(RELATIVE_PATH HA_PC_PREFIX_FROM_PC "${CMAKE_INSTALL_FULL_LIBDIR}/pkgconfig" "${CMAKE_INSTALL_PREFIX}")
file(RELATIVE_PATH HA_PC_LIBDIR "${CMAKE_INSTALL_PREFIX}" "${CMAKE_INSTALL_FULL_LIBDIR}")
file(RELATIVE_PATH HA_PC_INCLUDEDIR "${CMAKE_INSTALL_PREFIX}" "${CMAKE_INSTALL_FULL_INCLUDEDIR}")
configure_file(cmake/hail.pc.in ${CMAKE_CURRENT_BINARY_DIR}/hail.pc @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/hail.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
