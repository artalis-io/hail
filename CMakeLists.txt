if(HA_SANITIZER)
  if(MSVC)
    if(NOT CMAKE_C_COMPILER_ID STREQUAL "Clang" OR NOT HA_SANITIZER STREQUAL "address")
      message(FATAL_ERROR "MSVC mode supports only ClangCL HA_SANITIZER=address (ASan only)")
    endif()
    set(HA_SAN_FLAGS /fsanitize=address /Oy-)
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR
       NOT CMAKE_C_COMPILER_ARCHITECTURE_ID STREQUAL "x64" OR
       (CMAKE_MSVC_RUNTIME_LIBRARY AND NOT CMAKE_MSVC_RUNTIME_LIBRARY MATCHES "DLL"))
      message(FATAL_ERROR "ClangCL ASan qualification currently requires x64 and the DLL CRT")
    endif()
    execute_process(COMMAND ${CMAKE_C_COMPILER} /clang:-print-resource-dir
      OUTPUT_VARIABLE HA_CLANG_RESOURCES OUTPUT_STRIP_TRAILING_WHITESPACE
      RESULT_VARIABLE HA_CLANG_RESOURCE_RESULT)
    if(NOT HA_CLANG_RESOURCE_RESULT EQUAL 0)
      message(FATAL_ERROR "Cannot locate the ClangCL resource directory")
    endif()
    set(HA_ASAN_RUNTIME "${HA_CLANG_RESOURCES}/lib/windows/clang_rt.asan_dynamic-x86_64.lib")
    set(HA_ASAN_THUNK "${HA_CLANG_RESOURCES}/lib/windows/clang_rt.asan_dynamic_runtime_thunk-x86_64.lib")
    if(NOT EXISTS "${HA_ASAN_RUNTIME}" OR NOT EXISTS "${HA_ASAN_THUNK}")
      message(FATAL_ERROR "Install the matching ClangCL x64 ASan runtime")
    endif()
    target_compile_options(hail PUBLIC ${HA_SAN_FLAGS})
    target_link_libraries(hail PUBLIC "${HA_ASAN_RUNTIME}")
    target_link_options(hail PUBLIC /INCREMENTAL:NO /INCLUDE:__asan_seh_interceptor "/WHOLEARCHIVE:${HA_ASAN_THUNK}")
  elseif(HA_SANITIZER STREQUAL "address")
    set(HA_SAN_FLAGS -fsanitize=address,undefined -fno-omit-frame-pointer)
  elseif(HA_SANITIZER STREQUAL "thread")
    set(HA_SAN_FLAGS -fsanitize=thread -fno-omit-frame-pointer)
  elseif(HA_SANITIZER STREQUAL "undefined")
    set(HA_SAN_FLAGS -fsanitize=undefined -fno-omit-frame-pointer)
  else()
    message(FATAL_ERROR "Unknown HA_SANITIZER")
  endif()
  if(NOT MSVC)
    target_compile_options(hail PUBLIC ${HA_SAN_FLAGS})
    target_link_options(hail PUBLIC ${HA_SAN_FLAGS})
  endif()
endif()

  endif()

    message(FATAL_ERROR "Disable native backends for device-free fuzz builds")

  endif()

    message(FATAL_ERROR "Use a separate fuzzer build with HA_BUILD_TESTS=OFF and HA_BUILD_EXAMPLES=OFF")
  endif()

  foreach(harness config lifecycle queue faults)
    add_executable(hail_fuzz_${harness} fuzz/${harness}.c)
    target_compile_features(hail_fuzz_${harness} PRIVATE c_std_11)
    target_include_directories(hail_fuzz_${harness} PRIVATE src)
    target_compile_options(hail_fuzz_${harness} PRIVATE -Wall -Wextra -Wpedantic -Werror -fsanitize=fuzzer)
    target_link_options(hail_fuzz_${harness} PRIVATE -fsanitize=fuzzer)
    target_link_libraries(hail_fuzz_${harness} PRIVATE Hail::hail)
  endforeach()
endif()
