cmake_minimum_required(VERSION 3.16)
project(HailExternalDspConsumer LANGUAGES C)
# Independent project: resolve the installed package, never add Hail sources.
find_package(Hail 0.1.0 EXACT CONFIG REQUIRED)
add_executable(dsp_consumer main.c consumer.c)
target_link_libraries(dsp_consumer PRIVATE Hail::hail)
target_compile_features(dsp_consumer PRIVATE c_std_11)
if(MSVC)
  target_compile_options(dsp_consumer PRIVATE /W4 /WX)
else()
  target_compile_options(dsp_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow)
endif()
enable_testing()
add_test(NAME public_dsp_consumer COMMAND dsp_consumer)
set_tests_properties(public_dsp_consumer PROPERTIES TIMEOUT 180)
