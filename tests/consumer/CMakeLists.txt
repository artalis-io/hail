cmake_minimum_required(VERSION 3.16)
project(HailInstalledConsumer LANGUAGES C CXX)
find_package(Hail 0.1.0 EXACT CONFIG REQUIRED)
add_executable(consumer_c main.c)
target_compile_features(consumer_c PRIVATE c_std_11)
target_link_libraries(consumer_c PRIVATE Hail::hail)
add_executable(consumer_cpp ../headers.cpp)
target_link_libraries(consumer_cpp PRIVATE Hail::hail)
target_compile_features(consumer_cpp PRIVATE cxx_std_11)
enable_testing()
add_test(NAME installed_c COMMAND consumer_c)
add_test(NAME installed_cpp COMMAND consumer_cpp)
option(HA_CONSUMER_DSP "Exercise the independent public DSP fixture (requires virtual backend)" OFF)
if(HA_CONSUMER_DSP)
  add_subdirectory(../../examples/dsp-consumer dsp)
endif()
